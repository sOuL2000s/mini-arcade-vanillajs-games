<script>
    document.addEventListener('DOMContentLoaded', () => {
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const UI_SCORE = document.getElementById('scoreDisplay');
        const GAME_OVER_SCREEN = document.getElementById('gameOverScreen');
        const FINAL_SCORE = document.getElementById('finalScore');
        const RESTART_BUTTON = document.getElementById('restartButton');

        // --- Game Constants (These define the logic grid, fixed at 800x600) ---
        const CANVAS_WIDTH = 800;
        const CANVAS_HEIGHT = 600;
        const PLAYER_RADIUS = 10;
        const PROJECTILE_SPEED = 8;
        const PROJECTILE_SIZE = 2;

        // --- Game State ---
        let gameLoopId;
        let score = 0;
        let isGameOver = false;
        let keys = {};
        let mousePos = { x: 0, y: 0 };
        let lastUpdateTime = 0;
        let lastEnemySpawnTime = 0;
        let baseEnemySpawnInterval = 2000; // ms
        let enemies = [];
        let projectiles = [];
        let player = null;

        // --- Utility Functions ---

        /** Calculates distance between two points */
        function dist(x1, y1, x2, y2) {
            return Math.hypot(x2 - x1, y2 - y1);
        }

        /** Normalizes a vector */
        function normalize(dx, dy) {
            const length = Math.hypot(dx, dy);
            if (length === 0) return { x: 0, y: 0 };
            return { x: dx / length, y: dy / length };
        }

        // --- Entity Classes ---

        class Player {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.radius = PLAYER_RADIUS;
                this.speed = 4;
            }

            update() {
                if (isGameOver) return;

                let dx = 0;
                let dy = 0;

                // WASD or Arrow Keys
                if (keys['w'] || keys['ArrowUp']) dy -= 1;
                if (keys['s'] || keys['ArrowDown']) dy += 1;
                if (keys['a'] || keys['ArrowLeft']) dx -= 1;
                if (keys['d'] || keys['ArrowRight']) dx += 1;

                const vector = normalize(dx, dy);

                this.x += vector.x * this.speed;
                this.y += vector.y * this.speed;

                // Boundary checking
                this.x = Math.max(this.radius, Math.min(CANVAS_WIDTH - this.radius, this.x));
                this.y = Math.max(this.radius, Math.min(CANVAS_HEIGHT - this.radius, this.y));
            }

            draw() {
                ctx.fillStyle = '#00ffff';
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        class Enemy {
            constructor(x, y, levelMultiplier) {
                this.x = x;
                this.y = y;
                this.size = 15;
                // Base speed scales up steadily over time
                this.baseSpeed = 1.5;
                this.speed = this.baseSpeed + (levelMultiplier * 0.2); // Scaling factor
                this.color = '#ff4444';
                this.active = true;
            }

            // FSM: Chase State
            update() {
                if (!this.active || !player) return;

                const dx = player.x - this.x;
                const dy = player.y - this.y;

                const vector = normalize(dx, dy);

                this.x += vector.x * this.speed;
                this.y += vector.y * this.speed;

                // Check collision with player
                if (dist(this.x, this.y, player.x, player.y) < this.size / 2 + player.radius) {
                    gameOver();
                }
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.fillRect(this.x - this.size / 2, this.y - this.size / 2, this.size, this.size);
            }
        }

        class Projectile {
            constructor(x, y, velocityX, velocityY) {
                this.x = x;
                this.y = y;
                this.radius = PROJECTILE_SIZE;
                this.vx = velocityX;
                this.vy = velocityY;
                this.color = '#ffffff';
                this.active = true;
            }

            update() {
                if (!this.active) return;
                this.x += this.vx;
                this.y += this.vy;

                // Check boundaries
                if (this.x < 0 || this.x > CANVAS_WIDTH || this.y < 0 || this.y > CANVAS_HEIGHT) {
                    this.active = false;
                }
            }

            draw() {
                ctx.fillStyle = this.color;
                ctx.beginPath();
                ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        // --- Game Logic ---

        function initGame() {
            isGameOver = false;
            score = 0;
            UI_SCORE.textContent = 0;
            enemies = [];
            projectiles = [];
            lastEnemySpawnTime = 0;
            lastUpdateTime = performance.now();
            
            player = new Player(CANVAS_WIDTH / 2, CANVAS_HEIGHT / 2);
            GAME_OVER_SCREEN.style.display = 'none';

            if (gameLoopId) {
                cancelAnimationFrame(gameLoopId);
            }
            gameLoopId = requestAnimationFrame(gameLoop);
        }

        function spawnEnemy(levelMultiplier) {
            // Determine spawn point far outside the canvas edge
            let x, y;
            const buffer = 50;

            const edge = Math.floor(Math.random() * 4); // 0: Top, 1: Bottom, 2: Left, 3: Right

            switch (edge) {
                case 0: // Top
                    x = Math.random() * CANVAS_WIDTH;
                    y = -buffer;
                    break;
                case 1: // Bottom
                    x = Math.random() * CANVAS_WIDTH;
                    y = CANVAS_HEIGHT + buffer;
                    break;
                case 2: // Left
                    x = -buffer;
                    y = Math.random() * CANVAS_HEIGHT;
                    break;
                case 3: // Right
                    x = CANVAS_WIDTH + buffer;
                    y = Math.random() * CANVAS_HEIGHT;
                    break;
            }

            enemies.push(new Enemy(x, y, levelMultiplier));
        }

        function checkCollisions() {
            // 1. Projectile vs Enemy Collision
            projectiles.forEach(p => {
                if (!p.active) return;
                
                for (let i = enemies.length - 1; i >= 0; i--) {
                    const e = enemies[i];
                    if (!e.active) continue;

                    if (dist(p.x, p.y, e.x, e.y) < e.size / 2 + p.radius) {
                        // Collision! Destroy both.
                        p.active = false;
                        e.active = false;
                        score++;
                        UI_SCORE.textContent = score;
                        break; // Projectile is used up
                    }
                }
            });

            // Clean up inactive entities
            projectiles = projectiles.filter(p => p.active);
            enemies = enemies.filter(e => e.active);
        }
        
        function handleSpawningAndScaling(currentTime) {
            const timeElapsed = currentTime - lastUpdateTime;
            
            // Calculate level multiplier based on game time (seconds)
            const gameTimeSeconds = Math.floor(currentTime / 1000);
            const levelMultiplier = Math.floor(gameTimeSeconds / 10); // Increases every 10 seconds

            // Calculate current required spawn interval (scales down over time)
            // Minimum interval is 300ms
            const spawnInterval = Math.max(300, baseEnemySpawnInterval - (gameTimeSeconds * 50));

            if (currentTime - lastEnemySpawnTime > spawnInterval) {
                spawnEnemy(levelMultiplier);
                lastEnemySpawnTime = currentTime;
            }
        }

        function updateGame(currentTime) {
            if (isGameOver) return;

            handleSpawningAndScaling(currentTime);

            player.update();
            enemies.forEach(e => e.update());
            projectiles.forEach(p => p.update());

            checkCollisions();
        }

        function drawGame() {
            ctx.clearRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

            projectiles.forEach(p => p.draw());
            enemies.forEach(e => e.draw());
            player.draw();
        }

        function gameOver() {
            isGameOver = true;
            cancelAnimationFrame(gameLoopId);
            
            FINAL_SCORE.textContent = score;
            GAME_OVER_SCREEN.style.display = 'flex';
        }

        function gameLoop(currentTime) {
            lastUpdateTime = currentTime;

            updateGame(currentTime);
            drawGame();

            if (!isGameOver) {
                gameLoopId = requestAnimationFrame(gameLoop);
            }
        }

        // --- Event Listeners ---

        // Player shooting
        canvas.addEventListener('click', (event) => {
            if (isGameOver || !player) return;

            // Get mouse position relative to canvas
            const rect = canvas.getBoundingClientRect();
            
            // Scaling correction: Calculate the mouse position relative to the 800x600 logical grid
            const mouseX = (event.clientX - rect.left) * (CANVAS_WIDTH / rect.width);
            const mouseY = (event.clientY - rect.top) * (CANVAS_HEIGHT / rect.height);

            // Calculate direction vector from player to mouse
            const dx = mouseX - player.x;
            const dy = mouseY - player.y;

            const vector = normalize(dx, dy);

            // Create projectile, launching from player center
            const p = new Projectile(
                player.x + vector.x * (player.radius + 1), // Start just outside player edge
                player.y + vector.y * (player.radius + 1),
                vector.x * PROJECTILE_SPEED,
                vector.y * PROJECTILE_SPEED
            );
            projectiles.push(p);
        });

        // Key Press Handling for smooth movement
        document.addEventListener('keydown', (e) => {
            keys[e.key.toLowerCase()] = true;
        });

        document.addEventListener('keyup', (e) => {
            keys[e.key.toLowerCase()] = false;
        });

        // Restart Button
        RESTART_BUTTON.addEventListener('click', initGame);

        // Start the game initialization
        initGame();
    });
</script>

<script>
    // --- Configuration ---
    const CANVAS_WIDTH = 800;
    const CANVAS_HEIGHT = 600;

    // --- Game State Variables ---
    const canvas = document.getElementById('gameCanvas');
    const ctx = canvas.getContext('2d');
    // Canvas buffer sizes are fixed for consistent game logic
    canvas.width = CANVAS_WIDTH; 
    canvas.height = CANVAS_HEIGHT;

    let gameRunning = false;
    let startTime;
    let score = 0;
    let frameCounter = 0;
    let difficultyMultiplier = 1.0;

    const player = {
        x: CANVAS_WIDTH / 2,
        y: CANVAS_HEIGHT / 2,
        radius: 8,
        speed: 4,
        color: '#88C0D0'
    };

    let bullets = [];
    const bulletBaseRadius = 3;
    const bulletBaseSpeed = 3.5;
    const keys = {};

    // --- UI Elements ---
    const scoreDisplay = document.getElementById('score-display');
    const bulletCountDisplay = document.getElementById('bullet-count');
    const gameOverScreen = document.getElementById('game-over-screen');
    const finalScoreDisplay = document.getElementById('final-score');
    const finalDifficultyDisplay = document.getElementById('final-difficulty');
    const restartButton = document.getElementById('restart-button');

    // --- Input Handling ---
    document.addEventListener('keydown', (e) => {
        if (gameRunning) {
            keys[e.key] = true;
            // Prevent scrolling
            if (['w', 'a', 's', 'd', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) {
                e.preventDefault();
            }
        }
    });

    document.addEventListener('keyup', (e) => {
        if (gameRunning) {
            keys[e.key] = false;
        }
    });

    restartButton.addEventListener('click', () => {
        startGame();
    });

    // --- Core Game Functions ---

    function startGame() {
        // Reset state
        player.x = CANVAS_WIDTH / 2;
        player.y = CANVAS_HEIGHT / 2;
        bullets = [];
        score = 0;
        frameCounter = 0;
        difficultyMultiplier = 1.0;
        startTime = performance.now();

        gameOverScreen.style.display = 'none';
        gameRunning = true;

        // Start the main loop
        requestAnimationFrame(gameLoop);
    }

    function endGame() {
        gameRunning = false;
        const totalSurvivalTime = Math.floor(performance.now() - startTime);
        
        finalScoreDisplay.textContent = totalSurvivalTime;
        finalDifficultyDisplay.textContent = difficultyMultiplier.toFixed(2);
        
        gameOverScreen.style.display = 'flex';
        // Clear any lingering key presses so movement doesn't trigger on restart
        for (let key in keys) {
            keys[key] = false;
        }
    }

    // --- Update Logic ---

    function updatePlayer() {
        let dx = 0;
        let dy = 0;

        // Check WASD or Arrow Keys
        if (keys['w'] || keys['W'] || keys['ArrowUp']) dy -= player.speed;
        if (keys['s'] || keys['S'] || keys['ArrowDown']) dy += player.speed;
        if (keys['a'] || keys['A'] || keys['ArrowLeft']) dx -= player.speed;
        if (keys['d'] || keys['D'] || keys['ArrowRight']) dx += player.speed;

        // Diagonal movement normalization (prevents faster diagonal movement)
        if (dx !== 0 && dy !== 0) {
            const magnitude = Math.sqrt(dx * dx + dy * dy);
            dx = (dx / magnitude) * player.speed;
            dy = (dy / magnitude) * player.speed;
        }

        player.x += dx;
        player.y += dy;

        // Keep player within bounds
        player.x = Math.max(player.radius, Math.min(CANVAS_WIDTH - player.radius, player.x));
        player.y = Math.max(player.radius, Math.min(CANVAS_HEIGHT - player.radius, player.y));
    }

    function updateBullets() {
        // Optimize: Use standard loop for high performance
        let i = bullets.length;
        while (i--) {
            const bullet = bullets[i];
            
            bullet.x += bullet.dx;
            bullet.y += bullet.dy;

            // Collision Check
            const dist = Math.hypot(player.x - bullet.x, player.y - bullet.y);
            
            if (dist < player.radius + bullet.radius) {
                endGame();
                return; // Stop processing instantly on collision
            }

            // Remove bullets far outside the screen bounds
            const buffer = 50;
            if (bullet.x < -buffer || bullet.x > CANVAS_WIDTH + buffer ||
                bullet.y < -buffer || bullet.y > CANVAS_HEIGHT + buffer) 
            {
                bullets.splice(i, 1);
            }
        }
    }

    function updateDifficulty() {
        // Difficulty scales by 0.15 every 5 seconds
        const elapsedSeconds = (performance.now() - startTime) / 1000;
        difficultyMultiplier = 1.0 + (elapsedSeconds / 5) * 0.15;
    }

    // --- Bullet Spawning Functions ---

    function spawnBullet(startX, startY, angleRadians) {
        // Base speed scaled by difficulty
        const speed = bulletBaseSpeed * difficultyMultiplier;
        
        bullets.push({
            x: startX,
            y: startY,
            radius: bulletBaseRadius,
            dx: Math.cos(angleRadians) * speed,
            dy: Math.sin(angleRadians) * speed,
            color: '#FF4444' // Red bullets
        });
    }

    function spawnPatterns() {
        const center = { x: CANVAS_WIDTH / 2, y: CANVAS_HEIGHT / 2 };
        
        // Dynamic spawn rate (more bullets as difficulty increases)
        const spawnRate = Math.max(1, Math.floor(6 / difficultyMultiplier));

        if (frameCounter % spawnRate !== 0) return;

        // Pattern 1: Expanding Spiral (0 to 180 frames)
        if ((frameCounter % 360) < 180) {
            const spiralAngle = (frameCounter * 0.1) % (2 * Math.PI);
            
            for (let i = 0; i < 3; i++) {
                // Launch three bullets slightly spread out
                spawnBullet(center.x, center.y, spiralAngle + i * 0.1);
            }
        }
        // Pattern 2: Edge Waves / Target Streams (180 to 360 frames)
        else {
            const streamSpeedMod = 0.5 + difficultyMultiplier * 0.5;

            // Spawn from alternating corners
            let spawnX, spawnY;
            if (frameCounter % 4 === 0) {
                spawnX = 0; spawnY = 0;
            } else if (frameCounter % 4 === 1) {
                spawnX = CANVAS_WIDTH; spawnY = 0;
            } else if (frameCounter % 4 === 2) {
                spawnX = 0; spawnY = CANVAS_HEIGHT;
            } else {
                spawnX = CANVAS_WIDTH; spawnY = CANVAS_HEIGHT;
            }

            // Calculate angle directly towards the player's current position
            const angleToPlayer = Math.atan2(player.y - spawnY, player.x - spawnX);

            // Launch a stream aiming slightly ahead or behind based on time
            const wobble = Math.sin(frameCounter * 0.05) * 0.3;
            spawnBullet(spawnX, spawnY, angleToPlayer + wobble);
        }
        
    }

    // --- Drawing Functions ---

    function draw() {
        // Clear canvas
        ctx.fillStyle = '#000';
        ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

        // Draw Player
        ctx.beginPath();
        ctx.arc(player.x, player.y, player.radius, 0, Math.PI * 2);
        ctx.fillStyle = player.color;
        ctx.fill();
        
        // Draw Bullets
        ctx.fillStyle = '#FF4444'; 
        for (const bullet of bullets) {
            ctx.beginPath();
            ctx.arc(bullet.x, bullet.y, bullet.radius, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    function updateUI() {
        score = Math.floor(performance.now() - startTime);
        scoreDisplay.textContent = score;
        bulletCountDisplay.textContent = bullets.length;
    }

    // --- Main Loop ---

    function gameLoop(timestamp) {
        if (!gameRunning) return;

        frameCounter++;
        
        updatePlayer();
        updateDifficulty();
        spawnPatterns();
        updateBullets(); // This handles collision detection
        
        draw();
        updateUI();

        requestAnimationFrame(gameLoop);
    }

    // Initialize the game structure
    startGame();

</script>

        // --- GAME VARIABLES & STATE ---
        const GAME_AREA_HEIGHT = 250;

        const PLAYER_START_X = 50;
        const INITIAL_JUMP_VELOCITY = -13;
        const GRAVITY = 0.7;
        const DIFFICULTY_SCALING_TIME = 10000; // Increase speed every 10 seconds (10000 ms)
        const SPEED_INCREMENT = 1.5; // How much speed increases

        let gameSpeed = 6; // Initial speed (pixels per frame)
        let playerY = GAME_AREA_HEIGHT - GROUND_HEIGHT - PLAYER_SIZE;
        let velocityY = 0;
        let isJumping = false;
        let isGameOver = true;
        let obstacles = [];

        let score = 0;
        let lastFrameTime = Date.now();
        let lastSpeedIncreaseTime = Date.now();
        let animationFrameId;
        let obstacleSpawnTimeout;

        // --- DOM ELEMENTS ---
        const gameContainer = document.getElementById('game-container');
        const player = document.getElementById('player');
        const scoreDisplay = document.getElementById('score-display');
        const gameOverScreen = document.getElementById('game-over-screen');
        const finalScoreDisplay = document.getElementById('final-score');

        // --- CORE GAME LOOP FUNCTIONS ---

        function startGame() {
            // Reset state
            isGameOver = false;
            gameSpeed = 6;
            score = 0;
            obstacles = [];
            lastFrameTime = Date.now();
            lastSpeedIncreaseTime = Date.now();
            playerY = GAME_AREA_HEIGHT - GROUND_HEIGHT - PLAYER_SIZE;
            velocityY = 0;
            isJumping = false;

            // Clear existing obstacles from DOM
            document.querySelectorAll('.obstacle').forEach(o => o.remove());
            
            // Hide game over screen
            gameOverScreen.style.display = 'none';

            // Start game processes
            player.style.backgroundColor = '#ff4500';
            gameLoop();
            scheduleNextObstacle();
        }

        function gameLoop() {
            if (isGameOver) {
                cancelAnimationFrame(animationFrameId);
                return;
            }

            const now = Date.now();
            const deltaTime = now - lastFrameTime; // Time elapsed since last frame
            lastFrameTime = now;

            updateScore(deltaTime);
            checkSpeedScaling(now);
            updatePlayer();
            updateObstacles();
            checkCollision();

            // Request next frame
            animationFrameId = requestAnimationFrame(gameLoop);
        }

        // --- PLAYER MECHANICS ---

        function jump() {
            if (!isJumping && !isGameOver) {
                isJumping = true;
                velocityY = INITIAL_JUMP_VELOCITY;
            }
        }

        function updatePlayer() {
            // 1. Apply Gravity
            velocityY += GRAVITY;
            playerY += velocityY;

            // 2. Check Ground Collision (y-axis)
            const groundY = GAME_AREA_HEIGHT - GROUND_HEIGHT - PLAYER_SIZE;
            if (playerY >= groundY) {
                playerY = groundY;
                velocityY = 0;
                isJumping = false;
            }

            // 3. Update DOM position
            player.style.top = `${playerY}px`;
        }

        // --- OBSTACLE MECHANICS ---

        function spawnObstacle() {
            if (isGameOver) return;

            const minWidth = 15;
            const maxWidth = 50;
            const minHeight = 20;
            const maxHeight = 60;

            const width = Math.floor(Math.random() * (maxWidth - minWidth + 1)) + minWidth;
            const height = Math.floor(Math.random() * (maxHeight - minHeight + 1)) + minHeight;
            
            const obstacleEl = document.createElement('div');
            obstacleEl.className = 'obstacle';
            obstacleEl.style.width = `${width}px`;
            obstacleEl.style.height = `${height}px`;
            
            // Position obstacle at the right edge, sitting on the ground
            const startX = gameContainer.offsetWidth; 
            const startY = GAME_AREA_HEIGHT - GROUND_HEIGHT - height;

            obstacleEl.style.left = `${startX}px`;
            obstacleEl.style.top = `${startY}px`;

            gameContainer.appendChild(obstacleEl);

            // Store object data for physics
            obstacles.push({
                element: obstacleEl,
                x: startX,
                width: width,
                height: height
            });
            
            scheduleNextObstacle();
        }

        function scheduleNextObstacle() {
            // Randomize spawn time for unpredictable difficulty
            const minSpawnTime = 1000;
            const maxSpawnTime = 2500;
            const spawnDelay = Math.random() * (maxSpawnTime - minSpawnTime) + minSpawnTime;
            
            clearTimeout(obstacleSpawnTimeout);
            obstacleSpawnTimeout = setTimeout(spawnObstacle, spawnDelay / (gameSpeed / 6)); // Shorter delay as speed increases
        }

        function updateObstacles() {
            const newObstacles = [];
            
            for (let i = 0; i < obstacles.length; i++) {
                const obs = obstacles[i];
                
                // Move left based on current game speed
                obs.x -= gameSpeed;
                obs.element.style.left = `${obs.x}px`;

                // Check if obstacle is still on screen
                if (obs.x + obs.width > 0) {
                    newObstacles.push(obs);
                } else {
                    // Remove off-screen obstacles from the DOM
                    obs.element.remove();
                }
            }
            obstacles = newObstacles; // Update the active obstacle list
        }

        // --- GAME LOGIC & SCORING ---

        function checkSpeedScaling(now) {
            if (now - lastSpeedIncreaseTime > DIFFICULTY_SCALING_TIME) {
                gameSpeed += SPEED_INCREMENT;
                lastSpeedIncreaseTime = now;
                console.log(`Speed Increased! New speed: ${gameSpeed.toFixed(1)}`);
            }
        }

        function updateScore(deltaTime) {
            score += deltaTime / 1000; // Convert milliseconds to seconds
            scoreDisplay.textContent = `Score: ${score.toFixed(2)}s`;
        }

        function checkCollision() {
            const pX = PLAYER_START_X;
            const pY = playerY;
            const pW = PLAYER_SIZE;
            const pH = PLAYER_SIZE;

            for (const obs of obstacles) {
                const oX = obs.x;
                const oW = obs.width;
                const oH = obs.height;
                const oY = GAME_AREA_HEIGHT - GROUND_HEIGHT - oH;

                // Collision Detection: Bounding Box Check
                const overlapX = (pX < oX + oW) && (pX + pW > oX);
                const overlapY = (pY + pH > oY); // Player bottom is below obstacle top

                if (overlapX && overlapY) {
                    gameOver();
                    return;
                }
            }
        }

        function gameOver() {
            isGameOver = true;
            clearTimeout(obstacleSpawnTimeout);
            cancelAnimationFrame(animationFrameId);

            player.style.backgroundColor = '#990000'; // Change color on death
            
            finalScoreDisplay.textContent = `Time Survived: ${score.toFixed(2)}s`;
            gameOverScreen.style.display = 'flex';
        }

        function restartGame() {
            startGame();
        }

        // --- INITIALIZATION & EVENT LISTENERS ---

        function init() {
            // Event listener for space bar and screen click
            document.addEventListener('keydown', (e) => {
                if (e.code === 'Space') {
                    e.preventDefault(); // Prevent scrolling
                    jump();
                }
            });
            gameContainer.addEventListener('click', jump);

            // Start the game initially
            startGame();
        }

        // Wait for the DOM to be fully loaded before initializing
        window.onload = init;

            0: { id: 0, connections: [false, false, false, false], css: 'pipe-block' },
            1: { id: 1, connections: [true, false, true, false], css: 'pipe-straight' },
            2: { id: 2, connections: [true, true, false, false], css: 'pipe-corner' },
            3: { id: 3, connections: [true, true, true, false], css: 'pipe-t-junction' },
            4: { id: 4, connections: [true, true, true, true], css: 'pipe-cross' }

            { r: -1, c: 0, index: 0, opposite: 2 }, // North
            { r: 0, c: 1, index: 1, opposite: 3 },  // East
            { r: 1, c: 0, index: 2, opposite: 0 },  // South
            { r: 0, c: -1, index: 3, opposite: 1 }   // West

        // --- UTILITY FUNCTIONS ---

        /**
         * Rotates the connection array clockwise by R steps (0, 1, 2, 3).
         */
        function rotateConnections(connections, rotation) {
            if (rotation === 0) return connections;
            
            const rotated = [...connections];
            const steps = rotation % 4;

            for (let i = 0; i < steps; i++) {
                const last = rotated.pop();
                rotated.unshift(last);
            }
            return rotated;
        }

        /**
         * Get the current effective connections for a cell based on its type and rotation.
         */
        function getCurrentConnections(cell) {
            const baseConnections = PIPE_MAP[cell.type].connections;
            return rotateConnections(baseConnections, cell.rotation);
        }

            return Math.floor(Math.random() * max);

            return r >= 0 && r < GRID_SIZE && c >= 0 && c < GRID_SIZE;

        // --- GENERATION LOGIC ---
        function generatePuzzle() {
            grid = [];
            moves = 0;
            
            // 1. Initialize Grid
            for (let r = 0; r < GRID_SIZE; r++) {
                grid[r] = [];
                for (let c = 0; c < GRID_SIZE; c++) {
                    grid[r][c] = {

                        type: 0,
                        rotation: 0,
                        isPath: false,
                        initialPathRotation: 0
                    };
                }
            }

            // 2. Define Start and End Points (Force Start/End to be on opposite edges)
            const edgePairs = [
                [{r: 0, c: getRandomInt(GRID_SIZE)}, {r: GRID_SIZE - 1, c: getRandomInt(GRID_SIZE)}], // Top/Bottom
                [{r: getRandomInt(GRID_SIZE), c: 0}, {r: getRandomInt(GRID_SIZE), c: GRID_SIZE - 1}]  // Left/Right

            // 3. Generate the winding path using Randomized DFS
            while (currentR !== endCell.r || currentC !== endCell.c) {
                const isNearEnd = Math.abs(endCell.r - currentR) + Math.abs(endCell.c - currentC) <= 2;
                
                let potentialMoves = DIRS.map((dir, index) => ({ dir, index }))
                    .filter(({ dir }) => isInsideGrid(currentR + dir.r, currentC + dir.c));
                
                // Prioritize moving towards the end point if the path is long enough or near the end
                if (pathLength >= minPathLength || isNearEnd) {
                    potentialMoves.sort((a, b) => {
                        const distA = Math.hypot(endCell.r - (currentR + a.dir.r), endCell.c - (currentC + a.dir.c));
                        const distB = Math.hypot(endCell.r - (currentR + b.dir.r), endCell.c - (currentC + b.dir.c));
                        return distA - distB;
                    });

                let nextMove = null;
                for (const move of potentialMoves) {
                    const nextR = currentR + move.dir.r;
                    const nextC = currentC + move.dir.c;
                    
                    // Do not step on existing path segments unless we are reaching the final cell
                    if (grid[nextR][nextC].isPath && (nextR !== endCell.r || nextC !== endCell.c)) {
                        continue;
                    }

                    // Look ahead: Ensure the chosen move doesn't lead into a dead end immediately (optional, hard to implement fully)
                    
                    nextMove = move;

                const nextR = currentR + chosenDir.r;

                if (pathHistory.length > 0) {

                            break;

                    requiredExitDirIndex = i;
                    break;
                }
            }
            
            if (requiredExitDirIndex === -1) return false;

            let traceR = startR + DIRS[requiredExitDirIndex].r;
            let traceC = startC + DIRS[requiredExitDirIndex].c;
            let entryDir = DIRS[requiredExitDirIndex].opposite; 
            
            const traceQueue = [{ r: traceR, c: traceC, entryDir: entryDir }];
            visited.add(`${traceR},${traceC}`);
            pathCells.push(grid[traceR][traceC]);
            
            let successfulTrace = false;

            while (traceQueue.length > 0) {
                const { r, c, entryDir: currentEntryDir } = traceQueue.shift();
                const cell = grid[r][c];
                
                if (r === endCell.r && c === endCell.c) {
                    successfulTrace = true;
                    break; 
                }
                
                // If we hit a non-path piece, we stop the trace here
                if (!cell.isPath) break;

                const connections = getCurrentConnections(cell);

                // 1. Check if the current pipe accepts entry
                if (!connections[currentEntryDir]) {
                    break;
                }

                // 2. Find the single exit direction (for non-cross pieces)
                let exitDirIndex = -1;
                for (let i = 0; i < 4; i++) {
                    if (i !== currentEntryDir && connections[i]) {
                        exitDirIndex = i;
                        break;
                    }
                }

                if (exitDirIndex === -1) {
                    // Dead end or cross/T junction needs further validation
                    // For single-path puzzles, this means an incorrect rotation.
                    break;
                }

                const nextR = r + DIRS[exitDirIndex].r;
                const nextC = c + DIRS[exitDirIndex].c;
                
                if (isInsideGrid(nextR, nextC)) {
                    const nextCell = grid[nextR][nextC];
                    const requiredEntryIndex = DIRS[exitDirIndex].opposite;

                    // 3. Check if the next pipe accepts entry
                    const nextConnections = getCurrentConnections(nextCell);
                    
                    if (nextR === endCell.r && nextC === endCell.c) {
                        // Reached the end marker, successful trace if it links
                        successfulTrace = true;
                        break;
                    }
                    
                    if (nextConnections[requiredEntryIndex]) {
                        // Successful link, continue trace
                        if (!visited.has(`${nextR},${nextC}`)) {
                            visited.add(`${nextR},${nextC}`);
                            pathCells.push(nextCell);
                            traceQueue.push({ r: nextR, c: nextC, entryDir: requiredEntryIndex });
                        }
                    } else {
                        break; // Next pipe rotated wrong
                    }
                } else {
                    break; // Path goes off grid
                }
            }
            
            document.querySelectorAll('.cell').forEach(el => el.classList.remove('on-path'));

            if (successfulTrace) {
                score++;
                document.getElementById('score-display').textContent = score;
                document.getElementById('message').style.display = 'block';
                
                pathCells.forEach(cell => {
                    const el = document.getElementById(`cell-${cell.r}-${cell.c}`);

                    document.getElementById('message').style.display = 'none';

        // DOM Elements

        let startTime = 0;
        let gameRunning = false;
        let targetVisible = false;
        let timeoutHandle = null;

        // Constants for Random Delays
        const MIN_DELAY = 1000; // 1 second
        const MAX_DELAY = 3000; // 3 seconds

        /**
         * Reads the current dynamic size of the target based on CSS media queries.
         */
        function getTargetSize() {
             // We rely on getComputedStyle to read the real size set by CSS/media queries
            const style = window.getComputedStyle(targetEl);
            return parseInt(style.width);
        }

        /**
         * Initializes and starts a new game session.
         */
        function startGame() {
            if (gameRunning) return;
            
            gameRunning = true;
            score = 0;
            targetTimeLimit = 800;
            targetVisible = false;
            
            // Clear visual state
            gameAreaEl.classList.remove('failure');
            targetEl.style.opacity = 0;
            overlayEl.style.display = 'none';

            updateDisplay();
            setupNewRound();
        }

        /**
         * Sets up the waiting phase before the target appears.
         */
        function setupNewRound() {
            targetEl.style.opacity = 0;
            targetVisible = false;
            gameAreaEl.style.cursor = 'pointer';
            
            // Calculate random delay
            const delay = Math.random() * (MAX_DELAY - MIN_DELAY) + MIN_DELAY;
            
            timeoutHandle = setTimeout(showTarget, delay);
        }

        /**
         * Makes the target visible in a random location and starts the timer.
         */
        function showTarget() {
            if (!gameRunning) return;

            // 1. Set Random Position 
            const areaWidth = gameAreaEl.clientWidth;
            const areaHeight = gameAreaEl.clientHeight;
            const targetSize = getTargetSize();

            const maxX = areaWidth - targetSize;
            const maxY = areaHeight - targetSize;

            const randomX = Math.floor(Math.random() * maxX);
            const randomY = Math.floor(Math.random() * maxY);

            targetEl.style.left = `${randomX}px`;
            targetEl.style.top = `${randomY}px`;
            
            // 2. Set Random Color 
            const randomColor = '#' + Math.floor(Math.random()*16777215).toString(16).padStart(6, '0');
            targetEl.style.backgroundColor = randomColor;

            // 3. Display target and start timer
            targetEl.style.opacity = 1;
            gameAreaEl.style.cursor = 'default';
            targetVisible = true;

            // 4. Set the failure timer

    <script>
        // --- Configuration ---
        const CANVAS_SIZE = 400;
        const GRID_SIZE = 20; // 20x20 grid
        const CELL_SIZE = CANVAS_SIZE / GRID_SIZE; // 20px per cell
        const INITIAL_SPEED_MS = 150;
        const SPEED_INCREASE_INTERVAL = 20;
        const SPEED_DECREMENT = 10;

        // --- Game State Variables ---
        let canvas, ctx;
        let snake;
        let food;
        let dx, dy;
        let score;
        let gameLoopInterval;
        let currentSpeed;
        let isMoving;
        let gameRunning = false;

        // --- DOM Elements ---
        const scoreDisplay = document.getElementById('score');
        const speedDisplay = document.getElementById('speed');
        const gameOverScreen = document.getElementById('game-over-screen');
        const finalScoreDisplay = document.getElementById('final-score');

        // --- Initialization ---

        window.onload = function() {
            canvas = document.getElementById('game-board');
            ctx = canvas.getContext('2d');
            
            // Initial message set up
            gameOverScreen.style.display = 'flex';
            document.getElementById('final-score').innerText = 'N/A';
            document.querySelector('#game-over-screen h1').innerText = 'COILED DESTINY';
            document.querySelector('#game-over-screen p').innerText = 'Press Start to Play';
            document.querySelector('#game-over-screen button').innerText = 'Start Game';
        };

        function startGame() {
            if (gameRunning) return;
            
            // Reset state
            snake = [{x: 10, y: 10}, {x: 9, y: 10}];
            dx = 1;
            dy = 0;
            score = 0;
            currentSpeed = INITIAL_SPEED_MS;
            isMoving = true;
            gameRunning = true;

            scoreDisplay.innerText = score;
            speedDisplay.innerText = 1;
            gameOverScreen.style.display = 'none';

            spawnFood();
            
            startLoop();

            document.removeEventListener('keydown', changeDirection); 
            document.addEventListener('keydown', changeDirection);
        }

        function startLoop() {
            clearInterval(gameLoopInterval);
            gameLoopInterval = setInterval(gameLoop, currentSpeed);
        }

        // --- Core Drawing and Movement ---

        function drawGrid() {
            ctx.strokeStyle = '#020210';
            ctx.lineWidth = 0.5;
            for (let i = 0; i <= GRID_SIZE; i++) {
                // Vertical lines
                ctx.beginPath();
                ctx.moveTo(i * CELL_SIZE, 0);
                ctx.lineTo(i * CELL_SIZE, CANVAS_SIZE);
                ctx.stroke();

                // Horizontal lines
                ctx.beginPath();
                ctx.moveTo(0, i * CELL_SIZE);
                ctx.lineTo(CANVAS_SIZE, i * CELL_SIZE);
                ctx.stroke();
            }
        }

        function draw() {
            // Clear the board
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, CANVAS_SIZE, CANVAS_SIZE);
            
            drawGrid();

            // Draw Food
            ctx.fillStyle = 'red';
            ctx.fillRect(food.x * CELL_SIZE, food.y * CELL_SIZE, CELL_SIZE, CELL_SIZE);

            // Draw Snake
            snake.forEach((segment, index) => {
                ctx.fillStyle = index === 0 ? '#0F0' : '#090';
                ctx.fillRect(segment.x * CELL_SIZE, segment.y * CELL_SIZE, CELL_SIZE, CELL_SIZE);
                
                // Add border/style for defined segments
                ctx.strokeStyle = '#000';
                ctx.strokeRect(segment.x * CELL_SIZE, segment.y * CELL_SIZE, CELL_SIZE, CELL_SIZE);
            });
        }

        function moveSnake() {
            if (!isMoving) return;

            // 1. Calculate new head position
            const newHead = {
                x: snake[0].x + dx,
                y: snake[0].y + dy
            };

            // 2. Check for Game Over conditions (hitting walls or self)
            if (
                newHead.x < 0 || 
                newHead.x >= GRID_SIZE || 
                newHead.y < 0 || 
                newHead.y >= GRID_SIZE ||
                checkSelfCollision(newHead)
            ) {
                gameOver();
                return;
            }

            // 3. Move the snake: Add new head
            snake.unshift(newHead);

            // 4. Check for food consumption
            if (newHead.x === food.x && newHead.y === food.y) {
                score++;
                scoreDisplay.innerText = score;
                spawnFood();

                // Check for speed increase
                if (score > 0 && score % SPEED_INCREASE_INTERVAL === 0) {
                    currentSpeed = Math.max(50, currentSpeed - SPEED_DECREMENT);
                    
                    const level = 1 + Math.floor(score / SPEED_INCREASE_INTERVAL);
                    speedDisplay.innerText = level;
                    
                    startLoop();
                }
            } else {
                // Remove the tail if no food was eaten
                snake.pop();
            }
            
            isMoving = true;
        }

        function gameLoop() {
            moveSnake();
            draw();
        }

        // --- Helper Functions ---

        function getRandomCoordinate() {
            return Math.floor(Math.random() * GRID_SIZE);
        }

        function spawnFood() {
            let newFood;
            let onSnake;
            
            do {
                newFood = { x: getRandomCoordinate(), y: getRandomCoordinate() };
                onSnake = snake.some(segment => segment.x === newFood.x && segment.y === newFood.y);
            } while (onSnake);
            
            food = newFood;
        }

        function checkSelfCollision(head) {
            for (let i = 1; i < snake.length; i++) {
                if (head.x === snake[i].x && head.y === snake[i].y) {
                    return true;
                }
            }
            return false;
        }

        function changeDirection(event) {
            if (!gameRunning || !isMoving) return;

            const keyPressed = event.key;
            
            const goingUp = dy === -1;
            const goingDown = dy === 1;
            const goingLeft = dx === -1;
            const goingRight = dx === 1;

            let newDx = dx;
            let newDy = dy;

            // Use WASD or Arrow Keys
            if ((keyPressed === 'ArrowLeft' || keyPressed === 'a') && !goingRight) {
                newDx = -1; newDy = 0;
            } else if ((keyPressed === 'ArrowRight' || keyPressed === 'd') && !goingLeft) {
                newDx = 1; newDy = 0;
            } else if ((keyPressed === 'ArrowUp' || keyPressed === 'w') && !goingDown) {
                newDx = 0; newDy = -1;
            } else if ((keyPressed === 'ArrowDown' || keyPressed === 's') && !goingUp) {
                newDx = 0; newDy = 1;
            }

            if (newDx !== dx || newDy !== dy) {
                dx = newDx;
                dy = newDy;
                isMoving = false;
            }
        }
