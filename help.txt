### Summary of Changes Implemented:

1.  **Fullscreen/Dynamic Sizing:**
    *   **`index.html`:** The `GAMES` array has been updated to use a large common native resolution (1200x800) for all entries. This forces the `scaleGameFrame` function to always calculate a scale factor that maximizes the use of the viewport space, effectively making all games fullscreen within the arcade wrapper.
    *   **All Game Files:** Canvas or container dimensions (`CANVAS_WIDTH`, `CANVAS_HEIGHT`, or equivalent) are now initialized and updated dynamically using `window.innerWidth` and `window.innerHeight`, ensuring they fill the entire `iframe` window.
    *   **Snake Game (`snake-game.html`):** Now calculates the `CELL_SIZE` dynamically based on the smaller of the container's width or height, maintaining the 20x20 aspect ratio while scaling to the maximum possible size.

2.  **Standardized Menu System:**

    *   **Menu Logic:** Pause/Resume functionality, along with an `Escape` key listener, has been integrated into every game's main loop (or event handlers for non-looping games like the City Builder and Pipe Connect).

        MENU_OVERLAY.style.display = 'none';

        gameRunning = true;
        isGamePaused = false;

        requestAnimationFrame(gameLoop);

        gameRunning = false;
        isGamePaused = true;
        const totalSurvivalTime = Math.floor(performance.now() - startTime);
        
        finalScoreDisplay.textContent = totalSurvivalTime;
        finalDifficultyDisplay.textContent = difficultyMultiplier.toFixed(2);
        
        gameOverScreen.style.display = 'flex';
        for (let key in keys) {
            keys[key] = false;
        }
    }

    // --- Update Logic ---

    function updatePlayer() {
        let dx = 0;
        let dy = 0;

        // Check WASD or Arrow Keys
        if (keys['w'] || keys['W'] || keys['ArrowUp']) dy -= player.speed;
        if (keys['s'] || keys['S'] || keys['ArrowDown']) dy += player.speed;
        if (keys['a'] || keys['A'] || keys['ArrowLeft']) dx -= player.speed;
        if (keys['d'] || keys['D'] || keys['ArrowRight']) dx += player.speed;

        if (dx !== 0 && dy !== 0) {
            const magnitude = Math.sqrt(dx * dx + dy * dy);
            dx = (dx / magnitude) * player.speed;
            dy = (dy / magnitude) * player.speed;
        }

        player.x += dx;
        player.y += dy;

        // Keep player within bounds
        player.x = Math.max(player.radius, Math.min(CANVAS_WIDTH - player.radius, player.x));
        player.y = Math.max(player.radius, Math.min(CANVAS_HEIGHT - player.radius, player.y));
    }

    function updateBullets() {
        let i = bullets.length;
        while (i--) {
            const bullet = bullets[i];
            
            bullet.x += bullet.dx;
            bullet.y += bullet.dy;

            // Collision Check
            const dist = Math.hypot(player.x - bullet.x, player.y - bullet.y);
            
            if (dist < player.radius + bullet.radius) {
                endGame();
                return; 
            }

            // Remove bullets far outside the screen bounds
            const buffer = 50;
            if (bullet.x < -buffer || bullet.x > CANVAS_WIDTH + buffer ||
                bullet.y < -buffer || bullet.y > CANVAS_HEIGHT + buffer) 
            {
                bullets.splice(i, 1);
            }
        }
    }

    function updateDifficulty() {
        const elapsedSeconds = (performance.now() - startTime) / 1000;
        difficultyMultiplier = 1.0 + (elapsedSeconds / 5) * 0.15;
    }

    function spawnBullet(startX, startY, angleRadians) {
        const speed = bulletBaseSpeed * difficultyMultiplier;
        
        bullets.push({
            x: startX,
            y: startY,
            radius: bulletBaseRadius,
            dx: Math.cos(angleRadians) * speed,
            dy: Math.sin(angleRadians) * speed,
            color: '#FF4444' 
        });
    }

    function spawnPatterns() {
        const center = { x: CANVAS_WIDTH / 2, y: CANVAS_HEIGHT / 2 };
        
        const spawnRate = Math.max(1, Math.floor(6 / difficultyMultiplier));

        if (frameCounter % spawnRate !== 0) return;

        if ((frameCounter % 360) < 180) {
            const spiralAngle = (frameCounter * 0.1) % (2 * Math.PI);
            
            for (let i = 0; i < 3; i++) {
                spawnBullet(center.x, center.y, spiralAngle + i * 0.1);
            }
        }
        else {
            let spawnX, spawnY;
            if (frameCounter % 4 === 0) {
                spawnX = 0; spawnY = 0;
            } else if (frameCounter % 4 === 1) {
                spawnX = CANVAS_WIDTH; spawnY = 0;
            } else if (frameCounter % 4 === 2) {
                spawnX = 0; spawnY = CANVAS_HEIGHT;
            } else {
                spawnX = CANVAS_WIDTH; spawnY = CANVAS_HEIGHT;
            }

            const angleToPlayer = Math.atan2(player.y - spawnY, player.x - spawnX);

            const wobble = Math.sin(frameCounter * 0.05) * 0.3;
            spawnBullet(spawnX, spawnY, angleToPlayer + wobble);
        }
    }

    // --- Drawing Functions ---

    function draw() {
        ctx.fillStyle = '#000';
        ctx.fillRect(0, 0, CANVAS_WIDTH, CANVAS_HEIGHT);

        ctx.beginPath();
        ctx.arc(player.x, player.y, player.radius, 0, Math.PI * 2);
        ctx.fillStyle = player.color;
        ctx.fill();
        
        ctx.fillStyle = '#FF4444'; 
        for (const bullet of bullets) {
            ctx.beginPath();
            ctx.arc(bullet.x, bullet.y, bullet.radius, 0, Math.PI * 2);
            ctx.fill();
        }
    }

    function updateUI() {
        score = Math.floor(performance.now() - startTime);
        scoreDisplay.textContent = score;
        bulletCountDisplay.textContent = bullets.length;
    }

    // --- Main Loop ---

    function gameLoop(timestamp) {
        if (!gameRunning) return;
        
        if (!isGamePaused) { 
            frameCounter++;
            
            updatePlayer();
            updateDifficulty();
            spawnPatterns();
            updateBullets(); 
            updateUI();
        }
        
        draw();

        requestAnimationFrame(gameLoop);
    }

    // Initialize the game structure
    resizeCanvas();
    requestAnimationFrame(gameLoop); 

</script>

            if (keys['w'] || keys['arrowup']) dy -= moveStep;
            if (keys['s'] || keys['arrowdown']) dy += moveStep;
            if (keys['a'] || keys['arrowleft']) dx -= moveStep;
            if (keys['d'] || keys['arrowright']) dx += moveStep;

            if (dx !== 0 && dy !== 0) {
                const magnitude = Math.sqrt(dx * dx + dy * dy);
                dx = (dx / magnitude) * moveStep;
                dy = (dy / magnitude) * moveStep;
            }

            player.y += dy;

            const screenH = window.innerHeight;
            player.x = Math.max(player.width / 2, Math.min(screenW - player.width / 2, player.x));

            3: { id: 3, connections: [true, true, true, false], css: 'pipe-t-junction' },
            4: { id: 4, connections: [true, true, true, true], css: 'pipe-cross' }
        };

        const DIRS = [
            { r: -1, c: 0, index: 0, opposite: 2 }, // North
            { r: 0, c: 1, index: 1, opposite: 3 },  // East
            { r: 1, c: 0, index: 2, opposite: 0 },  // South
            { r: 0, c: -1, index: 3, opposite: 1 }   // West
        ];
        
        // Menu Elements
        const MENU_OVERLAY = document.getElementById('menu-overlay');
        const INSTRUCTIONS_BOX = document.getElementById('instructions-box');
        const BTN_RESUME = document.getElementById('btn-resume');
        const BTN_START = document.getElementById('btn-start');

        // --- Menu Logic ---
        function toggleInstructions() {

        // --- UTILITY FUNCTIONS ---

        function rotateConnections(connections, rotation) {
            if (rotation === 0) return connections;
            
            const rotated = [...connections];
            const steps = rotation % 4;

            for (let i = 0; i < steps; i++) {
                const last = rotated.pop();
                rotated.unshift(last);
            }
            return rotated;
        }

        function getCurrentConnections(cell) {
            const baseConnections = PIPE_MAP[cell.type].connections;
            return rotateConnections(baseConnections, cell.rotation);
        }

        function getRandomInt(max) {
            return Math.floor(Math.random() * max);
        }

        function isInsideGrid(r, c) {
            return r >= 0 && r < GRID_SIZE && c >= 0 && c < GRID_SIZE;
        }

        // --- GENERATION LOGIC ---
        function generatePuzzle() {
            // ... (Generation logic remains the same) ...
            grid = [];
            moves = 0;
            
            for (let r = 0; r < GRID_SIZE; r++) {
                grid[r] = [];
                for (let c = 0; c < GRID_SIZE; c++) {
                    grid[r][c] = {
                        r, c,
                        type: 0,
                        rotation: 0,
                        isPath: false,
                        initialPathRotation: 0
                    };
                }
            }

            const edgePairs = [
                [{r: 0, c: getRandomInt(GRID_SIZE)}, {r: GRID_SIZE - 1, c: getRandomInt(GRID_SIZE)}], 
                [{r: getRandomInt(GRID_SIZE), c: 0}, {r: getRandomInt(GRID_SIZE), c: GRID_SIZE - 1}] 
            ];
            
            const chosenPair = edgePairs[getRandomInt(2)];
            startCell = chosenPair[0];
            endCell = chosenPair[1];

            let currentR = startCell.r;
            let currentC = startCell.c;
            
            let pathLength = 0;
            grid[currentR][currentC].isPath = true;

            const pathHistory = [{ r: currentR, c: currentC }];
            const minPathLength = Math.floor(GRID_SIZE * 1.5);
            
            while (currentR !== endCell.r || currentC !== endCell.c) {
                const isNearEnd = Math.abs(endCell.r - currentR) + Math.abs(endCell.c - currentC) <= 2;
                
                let potentialMoves = DIRS.map((dir, index) => ({ dir, index }))
                    .filter(({ dir }) => isInsideGrid(currentR + dir.r, currentC + dir.c));
                
                if (pathLength >= minPathLength || isNearEnd) {
                    potentialMoves.sort((a, b) => {
                        const distA = Math.hypot(endCell.r - (currentR + a.dir.r), endCell.c - (currentC + a.dir.c));
                        const distB = Math.hypot(endCell.r - (currentR + b.dir.r), endCell.c - (currentC + b.dir.c));
                        return distA - distB;
                    });
                }
                
                let nextMove = null;
                for (const move of potentialMoves) {
                    const nextR = currentR + move.dir.r;
                    const nextC = currentC + move.dir.c;
                    
                    if (grid[nextR][nextC].isPath && (nextR !== endCell.r || nextC !== endCell.c)) {
                        continue;
                    }

                    nextMove = move;
                    break;
                }

                if (!nextMove) {
                    break;
                }

                const { dir: chosenDir, index: exitDirIndex } = nextMove;
                const nextR = currentR + chosenDir.r;
                const nextC = currentC + chosenDir.c;
                
                const prevCell = pathHistory[pathHistory.length - 1];
                let prevDirIndex = -1;
                
                if (pathHistory.length > 0) {
                    for(let i=0; i<4; i++) {
                        if (prevCell.r + DIRS[i].r === currentR && prevCell.c + DIRS[i].c === currentC) {
                            prevDirIndex = i;
                            break;
                        }
                    }
                }

                let pipeType;
                let finalRotation = 0;
                
                if (pathHistory.length === 1) { 
                    pipeType = 1; 
                    if (exitDirIndex === 1) finalRotation = 0; 
                    else if (exitDirIndex === 3) finalRotation = 2; 
                    else if (exitDirIndex === 0) finalRotation = 3; 
                    else if (exitDirIndex === 2) finalRotation = 1; 
                    
                } else if (nextR === endCell.r && nextC === endCell.c) { 
                    pipeType = 1;
                    if (prevDirIndex === 1) finalRotation = 0; 
                    else if (prevDirIndex === 3) finalRotation = 2; 
                    else if (prevDirIndex === 0) finalRotation = 3; 
                    else if (prevDirIndex === 2) finalRotation = 1; 
                } else {
                    pipeType = 2; 
                    let attempts = 0;
                    while(attempts < 4) {
                        const connections = rotateConnections(PIPE_MAP[2].connections, attempts);
                        if (connections[DIRS[prevDirIndex].opposite] && connections[exitDirIndex]) {
                            pipeType = 2;
                            finalRotation = attempts;
                            break;
                        }
                        attempts++;
                    }
                    
                    if (attempts === 4) {
                         pipeType = 3;
                         attempts = 0;
                         while(attempts < 4) {
                            const connections = rotateConnections(PIPE_MAP[3].connections, attempts);
                            if (connections[DIRS[prevDirIndex].opposite] && connections[exitDirIndex]) {
                                finalRotation = attempts;
                                break;
                            }
                            attempts++;
                        }
                    }
                }
                
                grid[currentR][currentC].type = pipeType;
                grid[currentR][currentC].initialPathRotation = finalRotation; 
                
                currentR = nextR;
                currentC = nextC;
                grid[currentR][currentC].isPath = true;
                pathHistory.push({ r: currentR, c: currentC });
                pathLength++;
            }
            
            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const cell = grid[r][c];
                    if (!cell.isPath) {
                        cell.type = getRandomInt(3) + 1; 
                        cell.initialPathRotation = getRandomInt(4);
                    }
                    
                    if ((r === startCell.r && c === startCell.c) || (r === endCell.r && c === endCell.c)) {
                         // Skip start/end markers
                    } else if(cell.isPath) {
                        cell.rotation = getRandomInt(4);
                    } else {
                        cell.rotation = getRandomInt(4);
                    }
                }
            }
            
            grid[startCell.r][startCell.c].type = 0; 
            grid[endCell.r][endCell.c].type = 0;

            renderGrid();
        }

        // --- VALIDATION LOGIC (Path Tracing) ---
        
        function checkWin() {
            if (isGamePaused) return false;
            // ... (rest of checkWin logic remains the same) ...
            
            const visited = new Set();
            const pathCells = [];
            
            const startR = startCell.r;
            const startC = startCell.c;
            
            let requiredExitDirIndex = -1;
            for(let i=0; i<DIRS.length; i++) {
                const neighborR = startR + DIRS[i].r;
                const neighborC = startC + DIRS[i].c;
                if (isInsideGrid(neighborR, neighborC) && grid[neighborR][neighborC].isPath && (neighborR !== endCell.r || neighborC !== endCell.c)) {
                    requiredExitDirIndex = i;
                    break;
                }
            }
            
            if (requiredExitDirIndex === -1) return false;

            let traceR = startR + DIRS[requiredExitDirIndex].r;
            let traceC = startC + DIRS[requiredExitDirIndex].c;
            let entryDir = DIRS[requiredExitDirIndex].opposite; 
            
            const traceQueue = [{ r: traceR, c: traceC, entryDir: entryDir }];
            visited.add(`${traceR},${traceC}`);
            pathCells.push(grid[traceR][traceC]);
            
            let successfulTrace = false;

            while (traceQueue.length > 0) {
                const { r, c, entryDir: currentEntryDir } = traceQueue.shift();
                const cell = grid[r][c];
                
                if (r === endCell.r && c === endCell.c) {
                    successfulTrace = true;
                    break; 
                }
                
                if (!cell.isPath) break;

                const connections = getCurrentConnections(cell);

                if (!connections[currentEntryDir]) {
                    break;
                }

                let exitDirIndex = -1;
                for (let i = 0; i < 4; i++) {
                    if (i !== currentEntryDir && connections[i]) {
                        exitDirIndex = i;
                        break;
                    }
                }

                if (exitDirIndex === -1) {
                    break;
                }

                const nextR = r + DIRS[exitDirIndex].r;
                const nextC = c + DIRS[exitDirIndex].c;
                
                if (isInsideGrid(nextR, nextC)) {
                    const nextCell = grid[nextR][nextC];
                    const requiredEntryIndex = DIRS[exitDirIndex].opposite;

                    const nextConnections = getCurrentConnections(nextCell);
                    
                    if (nextR === endCell.r && nextC === endCell.c) {
                        successfulTrace = true;
                        break;
                    }
                    
                    if (nextConnections[requiredEntryIndex]) {
                        if (!visited.has(`${nextR},${nextC}`)) {
                            visited.add(`${nextR},${nextC}`);
                            pathCells.push(nextCell);
                            traceQueue.push({ r: nextR, c: nextC, entryDir: requiredEntryIndex });
                        }
                    } else {
                        break; 
                    }
                } else {
                    break; 
                }
            }
            
            document.querySelectorAll('.cell').forEach(el => el.classList.remove('on-path'));

            if (successfulTrace) {
                score++;
                document.getElementById('score-display').textContent = score;
                document.getElementById('message').style.display = 'block';
                
                pathCells.forEach(cell => {
                    const el = document.getElementById(`cell-${cell.r}-${cell.c}`);
                    if (el) el.classList.add('on-path');
                });
                
                setTimeout(() => {
                    document.getElementById('message').style.display = 'none';
                    increaseDifficulty();
                    generatePuzzle();
                }, 1000); 
            }
            
            return successfulTrace;
        }

        // --- RENDER & INPUT ---

        function handleCellClick(r, c) {
            if (isGamePaused) return; 

            const cell = grid[r][c];
            if (cell.type === 0) return;
            
            cell.rotation = (cell.rotation + 1) % 4;
            moves++;
            document.getElementById('moves-display').textContent = moves;
            
            const element = document.getElementById(`cell-${r}-${c}`);
            if (element) {
                const pipeVisual = element.querySelector('.pipe-visual');
                pipeVisual.style.transform = `rotate(${cell.rotation * 90}deg)`;
                element.classList.remove('on-path'); 
            }

            checkWin();
        }

        function renderGrid() {
            GRID_CONTAINER.innerHTML = '';
            document.documentElement.style.setProperty('--grid-size', GRID_SIZE);
            document.getElementById('difficulty-display').textContent = `${GRID_SIZE}x${GRID_SIZE}`;
            
            document.getElementById('moves-display').textContent = 0;

            for (let r = 0; r < GRID_SIZE; r++) {
                for (let c = 0; c < GRID_SIZE; c++) {
                    const cellData = grid[r][c];
                    const cellDiv = document.createElement('div');
                    cellDiv.className = 'cell';
                    cellDiv.id = `cell-${r}-${c}`;
                    cellDiv.setAttribute('data-r', r);
                    cellDiv.setAttribute('data-c', c);
                    cellDiv.setAttribute('data-type', cellData.type);
                    
                    cellDiv.onclick = () => handleCellClick(r, c);

                    const visualDiv = document.createElement('div');
                    visualDiv.className = 'pipe-visual';
                    visualDiv.style.transform = `rotate(${cellData.rotation * 90}deg)`;
                    
                    cellDiv.appendChild(visualDiv);

                    if (r === startCell.r && c === startCell.c) {
                        cellDiv.classList.add('start');
                        cellDiv.textContent = 'S';
                    } else if (r === endCell.r && c === endCell.c) {
                        cellDiv.classList.add('end');
                        cellDiv.textContent = 'E';
                    }
                    
                    GRID_CONTAINER.appendChild(cellDiv);
                }
            }
        }
        
        // --- DIFFICULTY PROGRESSION ---

        function increaseDifficulty() {
            if (score % 3 === 0 && score > 0) {
                GRID_SIZE = Math.min(GRID_SIZE + 1, 12);
            }
        }

        // --- INITIALIZATION ---

        function initGame() {
            document.getElementById('score-display').textContent = score;
            document.getElementById('moves-display').textContent = moves;
            
            generatePuzzle();
            
            pauseGame(); 
            BTN_START.style.display = 'block';
            BTN_RESUME.style.display = 'none';
        }
        
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                isGamePaused ? resumeGame() : pauseGame();
            }
        });

        initGame();
    </script>
